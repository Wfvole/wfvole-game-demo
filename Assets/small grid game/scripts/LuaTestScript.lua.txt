---@diagnostic disable: undefined-global, undefined-field, need-check-nil

local speed = 10
---@type any
local lightCpnt = nil

function luaReadBuildingData(buildingdata)
    if self == nil or buildingdata == nil then
        return
    end

    local buildingName = buildingdata.buildingName
    if buildingName == nil or buildingName == "" then
        buildingName = buildingdata.name
    end

    if self.nameText ~= nil then
        self.nameText.text = buildingName or ""
    end

    if self.introductionText ~= nil then
        self.introductionText.text = buildingdata.introduction or ""
    end

    if self.produceText ~= nil then
        self.produceText.text = "产出" .. "光" .. string.format("%.1f", buildingdata.lightRate or 0) .. "/s  生命" .. string.format("%.1f", buildingdata.lifeRate or 0) .. "/s"
    end

    if self.consumeText ~= nil then
        self.consumeText.text = "消耗" .. tostring(buildingdata.useResourceType) .. string.format("%.1f", buildingdata.consume or 0) .. "/s"
    end
end
